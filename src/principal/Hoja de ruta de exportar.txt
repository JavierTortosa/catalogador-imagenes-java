8. Zoom en el modal (detalle implementación JS)
// zoom con rueda
modalImg.addEventListener('wheel', function(e) {
    e.preventDefault();
    let scale = this._scale || 1;
    scale += e.deltaY > 0 ? -0.1 : 0.1;
    scale = Math.max(0.5, Math.min(5, scale));
    this._scale = scale;
    this.style.transform = 'scale(' + scale + ')';
    this.style.transformOrigin = 'top left';
});

// pan con arrastre
modalImg.addEventListener('mousedown', function(e) {
    // iniciar arrastre
    let startX = e.clientX, startY = e.clientY;
    let origLeft = this._panX || 0, origTop = this._panY || 0;
    function onMove(e2) {
        this._panX = origLeft + (e2.clientX - startX);
        this._panY = origTop + (e2.clientY - startY);
        this.style.left = this._panX + 'px';
        this.style.top = this._panY + 'px';
    }
    // ... addEventListener mousemove/mouseup
});
Para móvil: eventos touch equivalentes + doble tap para toggle zoom 2×.
